# 2. 디렉토리 구조

```
dk_client/
├── basic/          # 플랫폼 추상화 유틸리티
├── Client/         # 메인 게임 클라이언트
│   ├── DXLib/      # DirectX → SDL2 호환 레이어 (Input/Sound/Music)
│   ├── framelib/   # 프레임 애니메이션 시스템
│   ├── JpegLib/    # JPEG 라이브러리 헤더
│   ├── OtherClass/ # 패킷 팩토리 매니저
│   ├── Packet/     # 네트워크 패킷 정의
│   │   ├── Cpackets/  # CG* 클라이언트→서버 패킷
│   │   ├── Gpackets/  # GC* 서버→클라이언트 패킷
│   │   ├── Lpackets/  # LC/LG 로그인 패킷
│   │   ├── Rpackets/  # 요청 패킷
│   │   ├── Upackets/  # 업데이트 패킷
│   │   └── Types/     # 공통 타입 정의
│   ├── SpriteLib/  # 스프라이트 렌더링 (SDL2 백엔드)
│   ├── SXml/       # XML 파싱
│   ├── TextSystem/ # 텍스트 렌더링 (SDL2 + freetype2)
│   ├── Updater/    # 게임 업데이터
│   └── WinLib/     # Windows 레거시 레이어
├── VS_UI/          # UI 프레임워크
│   ├── src/        # UI 구현 (위젯, 다이얼로그, 스킨)
│   ├── widget/     # 버튼, 스크롤바, 에디터 위젯
│   ├── hangul/     # 한글 IME 입력 지원
│   └── Imm/        # 진동 피드백 라이브러리
├── tools/          # 개발 도구 (뷰어, 리소스 검증)
└── 분석관련/        # 분석 문서
```
